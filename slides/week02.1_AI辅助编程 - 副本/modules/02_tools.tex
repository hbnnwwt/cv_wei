		\textbf{Ctrl+K：原地编辑}

			\item 按Ctrl+K

		\end{itemize}
	\end{columns}

		\item 输入指令："添加异常处理和日志"

	\textbf{场景2：理解项目代码}

		\item 按 \texttt{Ctrl+L} 打开聊天
		\item 输入："这个函数的作用是什么？"
		\item 使用 \texttt{@Files} 引用相关文件
		\item AI 结合上下文给出解释

	\vspace{0.3cm}

	\begin{exampleblock}{实战技巧}
		\begin{itemize}

			\item 不满意可以要求重写或解释修改原因
		\end{itemize}
	\end{exampleblock}
\end{frame}

\begin{frame}{Claude Code核心命令}
	\begin{center}
		\begin{tabular}{ll}
			\toprule
			\textbf{命令} & \textbf{功能} \\
			\midrule
			\texttt{claude} & 启动交互式对话 \\
			\texttt{claude "解释这个函数"} & 单次提问 \\
			\texttt{claude --file main.py} & 指定文件上下文 \\
			\texttt{claude --edit} & 编辑模式 \\
			\texttt{claude --help} & 查看帮助 \\
			\bottomrule
		\end{tabular}
	\end{center}

	\vspace{0.5cm}

	\textbf{典型使用流程：}
	\begin{enumerate}
		\item 打开终端，进入项目目录
		\item 运行 \texttt{claude} 启动对话
		\item 描述需求："帮我重构这个模块"
		\item AI 自动分析代码并给出修改建议
		\item 确认后自动应用修改
	\end{enumerate}

	\vspace{0.3cm}

	\begin{block}{高级技巧}
		\begin{itemize}
			\item 结合 Git 使用：AI可以理解提交历史
			\item 多文件重构：自动更新相关文件
			\item 生成测试：自动生成测试用例
		\end{itemize}
	\end{block}
\end{frame}

\begin{frame}{Claude Code实战演示}
	\textbf{场景：重构答题卡检测模块}
	
	\vspace{0.3cm}

	\begin{block}{实际对话示例}
		\small
		\$ cd answer-sheet-detector
		\\
		\$ claude
		\\
		You: 请帮我重构 detect\_edges.py，提高代码可读性和性能
		\\
		Claude: 我来分析这个文件... \\
		\indent{[分析当前代码结构]} \\
		\indent{[提出重构建议]} \\
		\indent{1. 将边缘检测逻辑封装为类} \\
		\indent{2. 添加类型提示和文档} \\
		\indent{3. 优化性能瓶颈}
		\\
		You: 同意，请执行重构
		\\
		Claude: 正在重构... [完成]
	\end{block}

	\vspace{0.3cm}

	\begin{alertblock}{注意}
		使用Claude Code需要Claude API Key，建议有经验的高级用户使用
	\end{alertblock}
