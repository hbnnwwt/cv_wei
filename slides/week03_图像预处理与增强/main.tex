\section*{课程概览}
\begin{frame}{课程概览}
    \begin{columns}
        \column{0.45\textwidth}
        \textbf{本周内容：}
        \begin{itemize}
            \item 图像处理流水线概述
            \item 图像去噪（滤波器）
            \item 图像增强（直方图、CLAHE）

            \item 判断题识别（Week 6）
            \item OCR文字识别（Week 7-8）

    \end{columns}
