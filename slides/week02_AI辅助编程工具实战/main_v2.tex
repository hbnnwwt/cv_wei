\section{互动环节（二）}
\subsection{代码找错挑战}
\begin{frame}[fragile]{代码找错挑战}
	\begin{block}{题目：下面的OpenCV代码有什么问题？}
		\lstset{language=Python}
		\begin{lstlisting}
import cv2
import numpy as np

img = cv2.imread('test.jpg')
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
edges = cv2.Canny(gray, 50, 150)

contours, hierarchy = cv2.findContours(
    edges, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

# 找最大轮廓
max_contour = max(contours, key=cv2.contourArea)

# 绘制轮廓
cv2.drawContours(img, [max_contour], -1, (0,255,0), 2)
cv2.imshow('Result', img)
cv2.waitKey(0)
cv2.destroyAllWindows()
		\end{lstlisting}
	\end{block}
\end{frame}

\begin{frame}[fragile]{代码找错挑战 - 答案}
	\begin{block}{问题分析}
		\begin{enumerate}
			\item \textbf{图像读取失败未检查}：如果文件不存在，img为None，后续会报错
			\item \textbf{资源未正确释放}：缺少异常时的资源清理
			\item \textbf{阈值固定}：Canny阈值50, 150是经验值，不适用于所有图像
			\item \textbf{轮廓为空未检查}：如果没找到轮廓，max()会报错
			\item \textbf{预处理缺失}：未进行降噪处理
		\end{enumerate}
	\end{block}
\end{frame}

\begin{frame}[fragile]{代码找错挑战 - 改进版}
	\lstset{language=Python}
	\begin{lstlisting}
import cv2
import numpy as np

# 1. 检查图像是否加载成功
img = cv2.imread('test.jpg')
if img is None:
    print("Error: Cannot load image")
    exit(1)

# 2. 预处理：降噪
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
blur = cv2.GaussianBlur(gray, (5,5), 0)

# 3. 自适应阈值
edges = cv2.Canny(blur, 50, 150)

# 4. 找轮廓（检查是否找到）
contours, _ = cv2.findContours(
    edges, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

if contours:  # 5. 检查轮廓是否为空
    max_contour = max(contours, key=cv2.contourArea)
    cv2.drawContours(img, [max_contour], -1, (0,255,0), 2)

cv2.imshow('Result', img)
cv2.waitKey(0)
cv2.destroyAllWindows()
	\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{代码找错挑战 - 思考题}
