		\begin{tikzpicture}
			\node[draw, fill=red!20, minimum width=3cm, minimum height=1.2cm] (ctrlk) at (0,0) {\shortstack{\textbf{Ctrl+K}\\原地编辑}};
			\node[draw, fill=green!20, minimum width=3cm, minimum height=1.2cm] (ctrll) at (4,0) {\shortstack{\textbf{Ctrl+L}\\上下文对话}};
			\node[draw, fill=blue!20, minimum width=3cm, minimum height=1.2cm] (ctrli) at (8,0) {\shortstack{\textbf{Ctrl+I}\\智能提示}};
		\end{tikzpicture}
	\end{center}

	\vspace{0.5cm}

	\begin{columns}
		\column{0.33\textwidth}
		\textbf{Ctrl+K：原地编辑}
		\begin{itemize}
			\item 选中代码
			\item 按Ctrl+K
			\item 输入修改指令
			\item AI原地修改
		\end{itemize}

		\column{0.33\textwidth}
		\textbf{Ctrl+L：上下文对话}
		\begin{itemize}
			\item 按Ctrl+L
			\item 打开聊天面板
			\item 多轮对话
			\item 引用文件上下文
		\end{itemize}

		\column{0.33\textwidth}
		\textbf{@引用功能}
		\begin{itemize}
			\item \texttt{@Files}：引用文件
			\item \texttt{@Code}：引用代码
			\item \texttt{@Docs}：引用文档
			\item \texttt{@Web}：搜索网络
		\end{itemize}
	\end{columns}
\end{frame}
