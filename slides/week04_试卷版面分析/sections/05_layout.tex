\begin{frame}{投影法原理}
    \textbf{什么是投影？}
    \begin{itemize}
        \item 统计图像在某个方向上的像素分布
        \item \textbf{水平投影}：统计每行的白色像素数
        \item \textbf{垂直投影}：统计每列的白色像素数
    \end{itemize}

    \vspace{0.3cm}

    \begin{columns}
        \column{0.5\textwidth}
        \textbf{水平投影应用：}
        \begin{itemize}
            \item 检测文本行分隔
            \item 找到题目之间的空白
            \item 切割试卷区域
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{垂直投影应用：}
        \begin{itemize}
            \item 检测字符分隔
            \item 找到列边界
            \item 对齐文本
        \end{itemize}
    \end{columns}
\end{frame}

def vertical_projection(binary):
    """

        \begin{itemize}
            \item 像素数多的位置
            \item 对应文本/图形区域
        \end{itemize}

        \textbf{波谷（Valley）：}
        \begin{itemize}
            \item 像素数少的位置
            \item 对应空白/分隔
        \end{itemize}
    \end{columns}

    \textbf{分隔线检测：} 找投影值低于阈值的波谷位置

\begin{frame}[fragile]{找到分隔线}
    \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
def find_divider_lines(proj, threshold=10, min_length=5):
    dividers, current_start = [], None
    for i, value in enumerate(proj):
        if value < threshold:
            if current_start is None:
                current_start = i
        else:
            if current_start is not None:
                length = i - current_start
                if length >= min_length:
                    dividers.append((current_start, i))
                current_start = None
    return dividers
dividers = find_divider_lines(h_proj, threshold=10)
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{投影法实战}
    \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
def layout_analysis(binary):
    h_proj = np.sum(binary // 255, axis=1)
    threshold = np.mean(h_proj) * 0.1
    dividers = find_divider_lines(h_proj, threshold)
    regions, prev_end = [], 0
    for start, end in dividers:
        if start > prev_end:
            regions.append((prev_end, start))
        prev_end = end
    if prev_end < binary.shape[0]:
        regions.append((prev_end, binary.shape[0]))
    return regions
regions = layout_analysis(binary)
print(f"Found {len(regions)} regions")
    \end{lstlisting}
\end{frame}

\begin{frame}{连通域分析}
    \textbf{什么是连通域？}
    \begin{itemize}
        \item 相邻的相同像素值构成的区域
        \item 4连通：上下左右
        \item 8连通：包括对角线
    \end{itemize}

    \vspace{0.3cm}

    \textbf{应用场景：}
    \begin{itemize}
        \item 识别填涂区域（选择题气泡）
        \item 分离独立字符
        \item 去除噪声斑点
    \end{itemize}
